\documentclass[tikz,border=2mm]{standalone}
\usetikzlibrary{arrows.meta,calc,shapes.geometric}
\begin{document}
\begin{tikzpicture}[line width=1.3pt, line cap=round, line join=round, x=1mm,y=1mm]
  % hot base
  \draw[thick] (-10,-10) -- (10,-10);
  % rising heat waves
  \foreach \x in {-6,0,6} {
    \draw[thick] (\x,-7) .. controls (\x-3,-3) and (\x+3,1) .. (\x,5)
                 .. controls (\x-1.5,7) and (\x-1,8) .. (\x,10);
  }
\end{tikzpicture}
