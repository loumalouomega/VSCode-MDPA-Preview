% Marketplace icon: shaded hexahedral mesh with a magnified triangular cell.
% Keep the canvas square and explicit so both PNG variants share padding.
% Render with `make icon` from icons/; the transparent variant retains the
% opaque lens and white separation around the magnifier for dark backgrounds.
\documentclass[tikz,border=0mm]{standalone}
\definecolor{meshfront}{HTML}{008C91}
\definecolor{meshtop}{HTML}{36BDB5}
\definecolor{meshside}{HTML}{00666F}
\definecolor{meshlight}{HTML}{D7F4EF}
\definecolor{meshmiddle}{HTML}{9DDED6}
\definecolor{meshdark}{HTML}{164E59}
\begin{document}
\begin{tikzpicture}[x=1mm,y=1mm,line cap=round,line join=round]
  \path[use as bounding box] (0,0) rectangle (72,72);
  \begin{scope}[shift={(0,6)},line width=0.85mm]
    % Three connected cells. Only exterior faces are visible; different
    % face values reveal the depth even when the icon is shown at 32 px.
    \fill[meshtop] (8,48) -- (17,57) -- (59,57) -- (50,48) -- cycle;
    \fill[meshfront] (8,27) -- (29,27) -- (29,6) -- (50,6)
      -- (50,48) -- (8,48) -- cycle;
    \fill[meshside] (50,6) -- (59,15) -- (59,57) -- (50,48) -- cycle;

    % Draw shared edges once, with consistent weight and rounded joins.
    \draw[white] (8,27) -- (8,48) -- (17,57) -- (59,57)
      -- (59,15) -- (50,6) -- (29,6) -- (29,27) -- cycle;
    \draw[white] (8,48) -- (50,48) -- (59,57);
    \draw[white] (29,27) -- (29,48) -- (38,57);
    \draw[white] (29,27) -- (50,27) -- (59,36);
    \draw[white] (50,48) -- (50,6) -- (29,27);
    \draw[white] (50,6) -- (59,36);

    % An opaque, pale lens isolates the enlarged mesh detail. White keylines
    % separate the dark ring and handle from the mesh on any background.
    \draw[white,line width=4mm] (57.8,8.2) -- (64.5,1.5);
    \draw[meshdark,line width=2.5mm] (57.8,8.2) -- (64.5,1.5);
    \fill[white] (49,17) circle[radius=13.5];
    \begin{scope}
      \clip (49,17) circle[radius=11.5];
      \fill[meshlight] (35,3) rectangle (63,31);
      \fill[meshmiddle] (35,25) -- (49,14) -- (63,25) -- (63,3) -- (35,3) -- cycle;
      \fill[meshtop] (49,14) -- (54,3) -- (35,3) -- (35,8) -- cycle;
      \draw[meshside,line width=0.65mm] (35,25) -- (49,14) -- (63,25);
      \draw[meshside,line width=0.65mm] (35,8) -- (49,14) -- (54,3);
      \fill[meshdark] (49,14) circle[radius=1.15];
    \end{scope}
    \draw[meshdark,line width=1.7mm] (49,17) circle[radius=11.5];
  \end{scope}
\end{tikzpicture}
\end{document}
